# Sub-proyecto nativo que ejecuta el recompilador cuando el proyecto principal es una compilacion cruzada
# (Emscripten, mingw...): las herramientas corren en la maquina que compila.  Lo lanza cmake/SF2Generation.cmake.
cmake_minimum_required(VERSION 3.20)
project(SF2Hostgen LANGUAGES C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_EXTENSIONS ON)
get_filename_component(SF2_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../.." ABSOLUTE)
set(CMAKE_SOURCE_DIR_ROOT "${SF2_ROOT}")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
list(APPEND CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/../../cmake")
set(SF2_HOSTGEN ON)
add_subdirectory("${CMAKE_SOURCE_DIR_ROOT}/third_party" third_party)
include(SF2Pipeline)
set(SF2_PARTS 16)
set(SF2_GEN_SOURCES "")
foreach(i RANGE 0 15)
  string(LENGTH "${i}" _len)
  if(_len EQUAL 1)
    set(_n "0${i}")
  else()
    set(_n "${i}")
  endif()
  list(APPEND SF2_GEN_SOURCES "recomp_${_n}.c")
endforeach()
list(APPEND SF2_GEN_SOURCES "recomp_table.c")
sf2_generation_pipeline("${CMAKE_BINARY_DIR}")
add_custom_target(all_generated ALL DEPENDS sf2_generate)
