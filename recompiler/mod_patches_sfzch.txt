# recompiler/mod_patches_sfzch.txt - parches de los mods de Street Fighter Zero (CPS Changer).  Mismo formato que mod_patches.txt.
# De los mods, sfzch tiene el sonido estéreo (D73), la pantalla ancha de presentación y la lógica ancha con el stage completo (D76).

# Lógica ancha y stage completo (D76).  Parches que dependen de c->mod_wide (píxeles extra por lado de la vista; 0 = juego original) y de
# c->mod_fixcam (stage completo: cámara fija en el centro del recorrido).  El 68000 de la placa lleva la cámara en el objeto 0xFF8280
# (a6): X de la vista (borde izquierdo de los 384 px originales) en 0x10(a6) = 0xFF8290, objetivo en 0x20(a6), topes en 0x44(a6) (izquierdo,
# 0x100 en todos los escenarios) y 0x46(a6) (derecho, 0x280).  La vista ancha mide 384 + 2 * mod_wide: los luchadores llegan hasta sus bordes
# y la cámara deja de acercarse al tope del arte lo que sobra de vista (topes + mod_wide y - mod_wide).
#   0x1DBA0  paredes de un luchador contra la cámara (a6 = luchador, d2 = semiancho): izquierda 0x1DBA6, derecha 0x1DBF6 (ancho 0x180)
0x1dba6 | uint32_t s = rd16(c, ea0); | uint32_t s = (rd16(c, ea0) - (uint32_t)c->mod_wide) & 0xffffu;
0x1dbf6 | uint32_t s = 0x180u; | uint32_t s = 0x180u + (uint32_t)c->mod_wide;
#   0x1DC40  empuje entre los dos luchadores contra las paredes (mismas cuentas)
0x1dc66 | uint32_t s = rd16(c, ea0); | uint32_t s = (rd16(c, ea0) - (uint32_t)c->mod_wide) & 0xffffu;
0x1dc8c | uint32_t s = 0x180u; | uint32_t s = 0x180u + (uint32_t)c->mod_wide;
#   0x191FC  seguimiento de la cámara: con los luchadores a menos de 0xC0 px la cámara solo se mueve si salen de la franja [0x60, 0x120]
#   de la vista; si no, se centra entre los dos.  La franja se ensancha con la vista (0x60 - e, 0x120 + e) y el umbral de separación con ella.
0x19204 | uint32_t s = 0xc0u; | uint32_t s = 0xc0u + 2u * (uint32_t)c->mod_wide;
0x19210 | uint32_t s = 0x120u; | uint32_t s = 0x120u + (uint32_t)c->mod_wide;
0x1921c | uint32_t s = 0x60u; | uint32_t s = (0x60u - (uint32_t)c->mod_wide) & 0xffffu;
#   0x19246 / 0x19264  topes de la cámara (0x46(a6) derecho, 0x44(a6) izquierdo).  Con c->mod_fixcam, o con una vista de 768 px o más, los dos
#   son el centro del recorrido y la cámara queda fija.

# Sonido estéreo (docs/ESTEREO.md): no cambia la partida.  La rutina 0x11350 encola el comando d1 en la ranura d0 de la cola de 256
# palabras -0x59E6(a5) = 0xFF261A (y detrás un 0xFF00, «sin comando»); quien llama deja en a6 el objeto que lo pide (medido con un tap
# de escritura sobre la cola en MAME).  Con c->mod_stereo != 0, mod_sound_tag (runtime/cpu.h, StereoRam de runtime/game.c) anota la
# X en pantalla de ese objeto.  Solo lee la RAM y escribe en campos del anfitrión (snd_x/snd_ok).
#   0x11364  move.w d1,0x0(a4,d0.w)
0x11364 | { uint32_t v = (c->d[1] & 0xffffu); | { if (c->mod_stereo) mod_sound_tag(c); uint32_t v = (c->d[1] & 0xffffu);
