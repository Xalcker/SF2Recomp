# recompiler/mod_patches_sfzch.txt - parches de los mods de Street Fighter Zero (CPS Changer).  Mismo formato que mod_patches.txt.
# De los mods, sfzch tiene el sonido estéreo (D73) y la pantalla ancha de presentación (D76); la lógica ancha y el stage completo
# necesitan las paredes y la cámara de SFZ, que no se han buscado.
#
# Sonido estéreo (docs/ESTEREO.md): no cambia la partida.  La rutina 0x11350 encola el comando d1 en la ranura d0 de la cola de 256
# palabras -0x59E6(a5) = 0xFF261A (y detrás un 0xFF00, «sin comando»); quien llama deja en a6 el objeto que lo pide (medido con un tap
# de escritura sobre la cola en MAME).  Con c->mod_stereo != 0, mod_sound_tag (runtime/cpu.h, StereoRam de runtime/game.c) anota la
# X en pantalla de ese objeto.  Solo lee la RAM y escribe en campos del anfitrión (snd_x/snd_ok).
#   0x11364  move.w d1,0x0(a4,d0.w)
0x11364 | { uint32_t v = (c->d[1] & 0xffffu); | { if (c->mod_stereo) mod_sound_tag(c); uint32_t v = (c->d[1] & 0xffffu);
#
# Pantalla ancha (D76, docs/WIDESCREEN.md): solo decide qué se dibuja y dónde, así que usa c->mod_view (píxeles extra por lado de la vista,
# en toda escena ancha; 0 = juego original).  En la máquina sombra (D62) corre con la lógica original.
#   0x13314/0x13318  ventana «¿está a la vista?» de los objetos (x - cámara + 0x40 <= 0x200): pone la bandera 0x1(a6) que decide si se dibujan

0x13318 | uint32_t s = 0x200u; | uint32_t s = 0x200u + 2u * (uint32_t)c->mod_view;

#   se vería: el recorte pasa a [-0x40 - mod_view, 0x1c0 + mod_view] (justo fuera de la vista) y la X se guarda con signo, sin máscara
#   (Video.xsigned; ModRam.signed_x).  Cinco puntos de recorte: comparación inferior, valor inferior, comparación superior, valor superior y máscara.
