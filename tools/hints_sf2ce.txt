offset_table 03c9cc 32 4

# Protección: 0x6538-0x6552 copia 0x51 bytes de 0x85EB8, sumando 3 a cada uno, en la RAM gráfica 0x910900 (dirección ofuscada,
# 0x488480 * 2), y 0x26F0-0x26FA la llama cada cuatro cuadros con `jsr (a0)`, a0 = 0x244240 << 2 (suma la RAM gráfica de
# 0x910A00-0x917FFF y escribe el resultado en 0x304(a5)...).  Se recompila en su dirección de RAM (D64).

# 0x666A-0x6686 copia 0x11 palabras de 0x72536, sumando 0x18 a cada una, en 0x918900 (0x246240 << 2); 0x3974-0x397E la llama con
# a0 = 0x123120 << 3 cuando el bit (cuadro & 15) de 0xE5 está a 0.
ram_code 918900 072536 22 0x18 w
